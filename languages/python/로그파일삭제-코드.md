# 로그파일삭제 코드

```bash
import os

def get_base_dir():
    """
    다양한 환경에서 실행될 수 있도록 기본 경로를 결정합니다.
    
    1. 환경 변수 'BASE_DIR'가 설정되어 있고 해당 경로가 존재하면 사용합니다.
    2. 그렇지 않으면 '/engine/datatrans' 경로가 존재하는지 확인하고, 존재하면 사용합니다.
    3. 위 조건에 모두 해당하지 않으면, 현재 스크립트 파일이 위치한 디렉토리를 기본 경로로 사용합니다.
    """
    # 1. 환경 변수 'BASE_DIR' 사용 (설정되어 있다면)
    env_base = os.environ.get("BASE_DIR")
    if env_base and os.path.exists(env_base):
        return env_base

    # 2. '/engine/datatrans' 경로가 존재하는 경우
    default_engine_path = "/engine/datatrans"
    if os.path.exists(default_engine_path):
        return default_engine_path

    # 3. 그 외에는 현재 스크립트 위치 기준으로 처리
    return os.path.dirname(os.path.abspath(__file__))

def cleanup_logs(folder_path):
    """
    지정된 폴더에서 파일 갯수를 확인한 후, 
    파일 갯수가 3개 이상인 경우 가장 오래된 파일부터 삭제하여 
    최신 2개 파일만 남도록 하는 함수입니다.
    """
    # 폴더 존재 여부 확인
    if not os.path.exists(folder_path):
        print(f"폴더가 존재하지 않습니다: {folder_path}")
        return

    # 폴더 내의 파일 목록 생성 (디렉토리는 제외)
    files = [os.path.join(folder_path, f) for f in os.listdir(folder_path)
             if os.path.isfile(os.path.join(folder_path, f))]
    
    file_count = len(files)
    print(f"폴더: {folder_path} - 파일 갯수: {file_count}")

    # 파일 갯수가 3개 이상인 경우에만 삭제 진행
    if file_count >= 3:
        # 파일들을 수정 시간 기준 오름차순 정렬 (가장 오래된 파일이 첫 번째)
        files_sorted = sorted(files, key=lambda x: os.path.getmtime(x))
        
        # 최신 파일 2개를 제외한 나머지 파일들을 삭제
        num_to_delete = file_count - 2
        print(f"삭제할 파일 수: {num_to_delete}")

        for file_path in files_sorted[:num_to_delete]:
            try:
                os.remove(file_path)
                print(f"삭제 완료: {file_path}")
            except Exception as e:
                print(f"파일 삭제 실패: {file_path} - 에러: {e}")
    else:
        print(f"삭제 조건 미충족 (파일 갯수가 3개 미만): {folder_path}")

def main():
    # 기본 경로 결정
    base_dir = get_base_dir()
    print(f"기본 경로: {base_dir}")

    # 첫번째 폴더: 기본 경로 아래 'DT_EDITOR/logs'
    folder1 = os.path.join(base_dir, "DT_EDITOR", "logs")
    # 두번째 폴더: 기본 경로 아래 'DT_SCHEDULE/logs'
    folder2 = os.path.join(base_dir, "DT_SCHEDULE", "logs")

    # 각 폴더에 대해 로그 파일 정리 실행
    cleanup_logs(folder1)
    cleanup_logs(folder2)

if __name__ == '__main__':
    main()

```

crontab을 등록하여 해당 python 파일을 실행시키고 싶은 시간에 실행한다

파이썬의 실행 경로에 따라 원하는 경로를 못찾을수도 있다 그 점을 고려하여 get_base_dir 함수 작성하였다
